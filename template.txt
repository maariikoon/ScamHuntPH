package com.scamhuntph.app

import android.content.Intent
import android.os.Bundle
import com.facebook.react.ReactActivity
import com.facebook.react.ReactActivityDelegate
import com.facebook.react.ReactRootView
import com.meedan.sharemenu.ShareMenuModule  // ✅ correct import

class ShareActivity : ReactActivity() {

  override fun onCreate(savedInstanceState: Bundle?) {
    super.onCreate(savedInstanceState)

    // Handle the incoming share intent
    val intent: Intent = intent
    ShareMenuModule.handleShareIntent(intent, this) // ✅ works now
  }

  override fun getMainComponentName(): String = "main"

  override fun createReactActivityDelegate(): ReactActivityDelegate {
    return object : ReactActivityDelegate(this, mainComponentName) {
      override fun createRootView(): ReactRootView {
        return ReactRootView(this@ShareActivity)
      }
    }
  }
}
